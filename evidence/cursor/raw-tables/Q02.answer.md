I'll look up Montgomery code violations for 2025 in Zetaris and rank census tracts by the official counts.The violation and tract layers are registered. I'll check whether the 2025 pages are complete, then join points to tract polygons.The five Montgomery County census tracts with the most 2025 code violations are:

| Tract | GEOID | Code violations |
|---|---|---|
| Census Tract 11 | 01101001100 | 1,125 |
| Census Tract 25 | 01101002500 | 955 |
| Census Tract 12 | 01101001200 | 818 |
| Census Tract 22.01 | 01101002201 | 765 |
| Census Tract 3 | 01101000300 | 702 |

These counts come from a Zetaris query that placed all 15,103 geocoded 2025 City of Montgomery code-violation points (ArcGIS `Code_Violations`, `Year = '2025'`; 336 of 15,439 records have no coordinates) inside Montgomery County census tracts (TIGERweb, FIPS 01101) with an even-odd point-in-polygon test; each point fell in exactly one tract.

```sql
WITH pts AS (
  SELECT f.attributes.OBJECTID AS objectid, f.geometry.x AS x, f.geometry.y AS y
  FROM (
    SELECT explode(features) AS f FROM MONTGOMERY_ARCGIS_REST.source_raw_p0
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p1
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p2
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p3
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p4
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p5
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p6
    UNION ALL SELECT explode(features) FROM MONTGOMERY_ARCGIS_REST.source_raw_p7
  )
  WHERE f.geometry.x IS NOT NULL AND f.geometry.y IS NOT NULL AND f.attributes.Year = '2025'
),
tract_feats AS (
  SELECT explode(features) AS f FROM CENSUS_TIGERWEB_REST.source_raw_p0
),
ring_pts AS (
  SELECT f.attributes.GEOID AS geoid, f.attributes.NAME AS name, ring_idx, pt_idx, pt[0] AS x, pt[1] AS y
  FROM tract_feats
  LATERAL VIEW posexplode(f.geometry.rings) r AS ring_idx, ring
  LATERAL VIEW posexplode(ring) p AS pt_idx, pt
),
edges AS (
  SELECT geoid, name, ring_idx, x AS x1, y AS y1,
         lead(x) OVER (PARTITION BY geoid, ring_idx ORDER BY pt_idx) AS x2,
         lead(y) OVER (PARTITION BY geoid, ring_idx ORDER BY pt_idx) AS y2
  FROM ring_pts
),
edges_f AS (
  SELECT * FROM edges WHERE x2 IS NOT NULL AND y1 <> y2
),
bbox AS (
  SELECT geoid, name,
         min(least(x1, x2)) AS minx, max(greatest(x1, x2)) AS maxx,
         min(least(y1, y2)) AS miny, max(greatest(y1, y2)) AS maxy
  FROM edges_f GROUP BY geoid, name
),
cand AS (
  SELECT p.objectid, p.x, p.y, b.geoid, b.name
  FROM pts p JOIN bbox b
    ON p.x >= b.minx AND p.x <= b.maxx AND p.y >= b.miny AND p.y <= b.maxy
),
hits AS (
  SELECT c.objectid, c.geoid, c.name,
         sum(CASE
           WHEN ((e.y1 > c.y) AND NOT (e.y2 > c.y) OR (e.y2 > c.y) AND NOT (e.y1 > c.y))
            AND (c.x < ((e.x2 - e.x1) * (c.y - e.y1) / (e.y2 - e.y1) + e.x1))
           THEN 1 ELSE 0 END) AS crossings
  FROM cand c JOIN edges_f e ON c.geoid = e.geoid
  GROUP BY c.objectid, c.geoid, c.name
),
inside AS (
  SELECT objectid, geoid, name FROM hits WHERE pmod(crossings, 2) = 1
)
SELECT geoid, name, count(*) AS violations
FROM inside GROUP BY geoid, name
ORDER BY violations DESC, geoid
LIMIT 5
```
